`Spike/PopoverSpike/SpikeAppDelegate.swift` — implements **both** Pattern 8 (NSPopover) AND a quick-toggle to Pattern 8a (sibling NSPanel) so we can A/B in the same session:
```swift
import AppKit

final class SpikeAppDelegate: NSObject, NSApplicationDelegate {
    private var widgetPanel: NSPanel!
    private var popover: NSPopover?
    private var siblingPopoverPanel: NSPanel?
    private var trackingArea: NSTrackingArea?

    func applicationDidFinishLaunching(_ n: Notification) {
        // 1. Spawn the widget panel exactly as RESEARCH Pattern 7 specifies
        widgetPanel = NSPanel(contentRect: NSRect(x: 1300, y: 800, width: 44, height: 44),
                              styleMask: [.borderless, .nonactivatingPanel],
                              backing: .buffered, defer: false)
        widgetPanel.level = .floating
        widgetPanel.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary, .stationary]
        widgetPanel.hasShadow = true
        widgetPanel.backgroundColor = .systemBlue
        widgetPanel.isOpaque = false
        widgetPanel.acceptsMouseMovedEvents = true

        let view = NSView(frame: NSRect(x: 0, y: 0, width: 44, height: 44))
        view.wantsLayer = true
        view.layer?.backgroundColor = NSColor.systemBlue.cgColor
        widgetPanel.contentView = view
        installTracking(on: view)
        widgetPanel.orderFront(nil)

        // Status print so the human running the spike knows what to expect
        print("Spike running. Hover the blue square. Then press ⌘-Tab — does this app appear?")
        print("Send SIGINT (Ctrl-C) to toggle: NSPopover (default) ↔ sibling-NSPanel (after first SIGINT).")
        signal(SIGINT) { _ in TogglesHolder.usePopover.toggle(); print("Topology now: \(TogglesHolder.usePopover ? "NSPopover" : "sibling-NSPanel")") }
    }

    private func installTracking(on view: NSView) {
        let opts: NSTrackingArea.Options = [.mouseEnteredAndExited, .activeAlways, .inVisibleRect]
        let ta = NSTrackingArea(rect: .zero, options: opts, owner: self, userInfo: nil)
        view.addTrackingArea(ta)
        self.trackingArea = ta
    }

    override func mouseEntered(with event: NSEvent) {
        if TogglesHolder.usePopover { showPopover() } else { showSiblingPanel() }
    }
    override func mouseExited(with event: NSEvent) {
        DispatchQueue.main.asyncAfter(deadline: .now() + .milliseconds(500)) { [weak self] in
            self?.popover?.performClose(nil)
            self?.siblingPopoverPanel?.orderOut(nil)
        }
    }

    private func showPopover() {
        let pop = NSPopover()
        pop.behavior = .transient
        let vc = NSViewController()
        let v = NSView(frame: NSRect(x: 0, y: 0, width: 280, height: 120))
        v.wantsLayer = true
        v.layer?.backgroundColor = NSColor.windowBackgroundColor.cgColor
        let label = NSTextField(labelWithString: "POPOVER (Pattern 8)")
        label.frame = NSRect(x: 12, y: 50, width: 256, height: 20)
        v.addSubview(label)
        vc.view = v
        pop.contentViewController = vc
        pop.show(relativeTo: NSRect.zero,
                 of: widgetPanel.contentView!,
                 preferredEdge: .minY)
        self.popover = pop
    }

    private func showSiblingPanel() {
        let p = NSPanel(contentRect: NSRect(x: 1000, y: 700, width: 280, height: 120),
                        styleMask: [.borderless, .nonactivatingPanel],
                        backing: .buffered, defer: false)
        p.level = .floating
        p.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary, .stationary]
        p.hasShadow = true
        p.isOpaque = true
        p.backgroundColor = .windowBackgroundColor
        let v = NSView(frame: NSRect(x: 0, y: 0, width: 280, height: 120))
        let label = NSTextField(labelWithString: "SIBLING PANEL (Pattern 8a)")
        label.frame = NSRect(x: 12, y: 50, width: 256, height: 20)
        v.addSubview(label)
        p.contentView = v
        p.orderFront(nil)
        self.siblingPopoverPanel = p
    }
}

enum TogglesHolder { static var usePopover = true }
```
