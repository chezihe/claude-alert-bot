```swift
// App/SoundPlayer.swift — Phase 2 AUD-01.
// RESEARCH Pattern 10 (lines 801-832). NSSound rejected per macOS 26 CoreAudio crash report.
// D2-19 — AVAudioPlayer direct. UNNotificationSound channel deliberately not used (auto-dismiss risk).
// Discretion default: /System/Library/Sounds/Funk.aiff (system-resident, no asset shipping).
import Foundation
import AVFoundation
import os

@MainActor
final class SoundPlayer {
    private let log = Logger(subsystem: "com.claudealert.bot.hook", category: "notification")
    private var player: AVAudioPlayer?

    /// Default = system Funk.aiff. Test/dev can pass a custom URL.
    init(soundURL: URL = URL(fileURLWithPath: "/System/Library/Sounds/Funk.aiff")) {
        do {
            let p = try AVAudioPlayer(contentsOf: soundURL)
            p.prepareToPlay()
            self.player = p
        } catch {
            self.player = nil
            log.error("SoundPlayer load failed at \(soundURL.path, privacy: .public): \(String(describing: error), privacy: .public)")
        }
    }

    /// Plays once. If a previous play is still in progress, restart from 0 (matches RESEARCH Pattern 10).
    /// Idempotent if player is nil (load failed) — silent no-op.
    func playOnce() {
        guard let p = player else { return }
        p.currentTime = 0
        p.play()
    }

    #if DEBUG
    var playerForTesting: AVAudioPlayer? { player }
    #endif
}
```

```swift
// App/NotificationOrchestrator.swift — Phase 2 MainActor orchestrator.
// PATTERNS.md §NotificationOrchestrator: AppDelegate component-wiring style.
// RESEARCH §"Architectural Responsibility Map" line 144: this class is the
// MainActor handoff point between SessionRegistry actor and AppKit widget.
// AUD-02: SettingsStore.soundEnabled gates SoundPlayer.playOnce.
// D2-18 RETRACTED: NO Focus/DnD detection — sound is solely gated by user toggle.
import Foundation
import os

@MainActor protocol WidgetControllerProtocol: AnyObject {
    func showWidget(pendingCount: Int, latest: CompletedSession?)
    func hideWidget()
    func updatePendingCount(_ n: Int, latest: CompletedSession?)
    func setQueue(_ queue: [CompletedSession])
}

@MainActor
final class NotificationOrchestrator: NotifierProtocol {
    private let log = Logger(subsystem: "com.claudealert.bot.hook", category: "notification")
    private weak var widget: (any WidgetControllerProtocol)?
    private let sound: SoundPlayer
    private let settings: () -> SettingsStore   // closure so tests can swap

    init(widget: (any WidgetControllerProtocol)?,
         sound: SoundPlayer = SoundPlayer(),
         settings: @autoclosure @escaping () -> SettingsStore = SettingsStore.shared) {
        self.widget = widget
        self.sound = sound
        self.settings = settings
    }

    /// Bind a widget controller after construction (Wave 6 wiring).
    func bind(widget: any WidgetControllerProtocol) {
        self.widget = widget
    }

    // MARK: - NotifierProtocol

    func present(session: CompletedSession, playSoundOnce: Bool) async {
        let store = settings()
        let count = (widget?.queueCountForLogging ?? 0) + 1
        widget?.showWidget(pendingCount: count, latest: session)
        if playSoundOnce && store.soundEnabled {
            sound.playOnce()
            log.notice("present session=\(session.sessionID, privacy: .public) sound=on count=\(count)")
        } else {
            log.notice("present session=\(session.sessionID, privacy: .public) sound=off count=\(count)")
        }
    }

    func refreshQueueState(completed: [CompletedSession], count: Int) async {
        if count == 0 {
            widget?.hideWidget()
            log.notice("refreshQueueState: queue empty → hideWidget")
        } else {
            widget?.setQueue(completed)
            widget?.updatePendingCount(count, latest: completed.last)
            log.notice("refreshQueueState: count=\(count)")
        }
    }
}

/// Convenience for logging only — not part of the protocol contract.
private extension WidgetControllerProtocol {
    var queueCountForLogging: Int { 0 }
}
```
