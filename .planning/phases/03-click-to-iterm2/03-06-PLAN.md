**Step 4 — Animation helper** (add as private method on PopoverRowView):

```swift
private func runMissingAnimation() {
    let reduceMotion = NSWorkspace.shared.accessibilityDisplayShouldReduceMotion
    if reduceMotion {
        // A11y: skip rotation, immediate collapse.
        withAnimation(.easeInOut(duration: 0.2)) {
            collapsed = true
            faded = true
        } completion: {
            onMissingComplete()
        }
        return
    }

    // Phase 1 (0..0.3s): 도리도리 — rotation +12 → -12 → 0 (1 round-trip).
    withAnimation(.easeInOut(duration: 0.15)) { rotation = 12 }
    DispatchQueue.main.asyncAfter(deadline: .now() + 0.15) {
        withAnimation(.easeInOut(duration: 0.15)) { rotation = -12 }
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.15) {
            // settle to 0 (no animation — instant) and chain to phase 2.
            rotation = 0
            // Phase 2 (0.3..0.7s): collapse + fade.
            withAnimation(.easeInOut(duration: 0.4)) {
                collapsed = true
                faded = true
            } completion: {
                onMissingComplete()
            }
        }
    }
}
```

`withAnimation(_:completionCriteria:_:completion:)` is available macOS 14+ (RESEARCH §"Standard Stack" cites Apple docs verbatim). Default completionCriteria is `.logicallyComplete` — matches what we need (callback fires when the animation logically finishes). If the executor encounters compiler issues with the omitted completionCriteria, use `withAnimation(.easeInOut(duration: 0.4), completionCriteria: .removed) { … } completion: { … }` per RESEARCH §Pattern 3 alt orchestration note.

    func test_RowState_caseIdentity() {
        // D3-11: exactly 3 cases. Brittle-by-design — bumping cases requires CONTEXT amendment.
        let normal: RowState = .normal
        let jumping: RowState = .jumping
        let missing: RowState = .missing
        XCTAssertNotEqual(normal, jumping)
        XCTAssertNotEqual(jumping, missing)
        XCTAssertNotEqual(normal, missing)

        // Lightweight contract test — the row's Button action checks `state == .normal`.
        // We assert the contract by inspecting that the existing test infrastructure

        // Surrogate: documentation test — assert the source contains the guard.

    func test_missingAnimation_callsOnMissingCompleteCallback() async {
        // Assert: when state transitions to .missing, runMissingAnimation eventually
        // calls onMissingComplete. We can't unit-test the SwiftUI .onChange wiring;
        // we can integration-test the callback dispatch by running the row in a host

        let src = try? String(contentsOfFile: "App/PopoverRowView.swift", encoding: .utf8)
