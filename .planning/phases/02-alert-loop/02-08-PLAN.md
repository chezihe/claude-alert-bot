// App/PopoverContentView.swift — declared at top of file or a separate WidgetContentRules file.
enum PopoverContentRules {
    static func shouldShowClearAll(rowCount: Int) -> Bool { rowCount >= 2 }

    /// D2-06: same-project duplicates → show time suffix on those rows only.
    static func projectsWithDuplicates(_ queue: [CompletedSession]) -> Set<String> {
        var counts: [String: Int] = [:]
        for s in queue { counts[s.projectName, default: 0] += 1 }
        return Set(counts.filter { $0.value >= 2 }.keys)
    }

    static func timeSuffix(for date: Date) -> String {
        let f = DateFormatter()
        f.dateFormat = "HH:mm"
        return f.string(from: date)
    }

    static func showsOrphanIndicator(session: CompletedSession) -> Bool {
        session.durationSec == nil
    }
}
```

// App/PopoverContentView.swift — Phase 2 popover container view.
// UI-SPEC §"Hover Popover" — 280pt width, max 8 visible rows + ScrollView, top-trailing Clear all.
// D2-07: Clear all visible when row count >= 2; non-destructive (no confirm).
import SwiftUI

struct PopoverContentView: View {
    let queue: [CompletedSession]
    let onRowClick: (String) -> Void   // sessionID
    let onClearAll: () -> Void

    private var dupProjects: Set<String> {
        PopoverContentRules.projectsWithDuplicates(queue)
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 0) {
            if PopoverContentRules.shouldShowClearAll(rowCount: queue.count) {
                HStack {
                    Spacer()
                    Button("모두 지우기", action: onClearAll)
                        .buttonStyle(.plain)
                        .font(.system(size: 11))
                        .foregroundStyle(Color(NSColor.secondaryLabelColor))
                        .accessibilityLabel("모든 알림 지우기")
                }
                .padding(.horizontal, 12)
                .padding(.top, 8)
            }
            ScrollView {
                VStack(spacing: 0) {
                    ForEach(queue) { session in
                        PopoverRowView(
                            session: session,
                            showTimeSuffix: dupProjects.contains(session.projectName),
                            onClick: { onRowClick(session.sessionID) }
                        )
                    }
                }
            }
            .scrollIndicators(.hidden)
            .frame(maxHeight: 36 * 8)   // max 8 visible rows
        }
        .frame(width: 280)
        .background(.thinMaterial)
    }
}

// App/WidgetPopoverController.swift — Phase 2 popover surface.
// Topology branch per 02-01-SPIKE-RESULT.md: Pattern 8 (NSPopover) or Pattern 8a (sibling NSPanel).
// Conforms to WidgetHoverDelegate (declared in 02-07's FloatingWidgetWindowController.swift).
// Hover-intent: 150ms entry delay, 250ms exit grace (UI-SPEC §"Floating Widget" hover row).
// Click: Task { @MainActor in await SessionRegistry.shared.clearOne(...) } + OSLog [would-jump session=<uuid>] (D2-08).
import AppKit
import SwiftUI
import os

@MainActor
final class WidgetPopoverController: NSObject, WidgetHoverDelegate {
    private let log = Logger(subsystem: "com.claudealert.bot.hook", category: "widget")
    private weak var widgetController: FloatingWidgetWindowController?
    private var entryWorkItem: DispatchWorkItem?
    private var exitWorkItem: DispatchWorkItem?

    // Pattern 8 path — NSPopover. If 02-01 verdict was 8a, replace with `private var popoverPanel: FloatingWidgetPanel?`.
    private var popover: NSPopover?

    init(widgetController: FloatingWidgetWindowController) {
        self.widgetController = widgetController
        super.init()
    }

    // MARK: - WidgetHoverDelegate

    func widgetMouseEntered() {
        // Cancel pending exit (UI-SPEC: re-entering during grace cancels dismiss)
        exitWorkItem?.cancel(); exitWorkItem = nil
        let intent = DispatchWorkItem { [weak self] in self?.showPopover() }
        entryWorkItem = intent
        DispatchQueue.main.asyncAfter(deadline: .now() + .milliseconds(150), execute: intent)
    }

    func widgetMouseExited() {
        // Cancel pending entry
        entryWorkItem?.cancel(); entryWorkItem = nil
        let exit = DispatchWorkItem { [weak self] in self?.dismissPopover() }
        exitWorkItem = exit
        DispatchQueue.main.asyncAfter(deadline: .now() + .milliseconds(250), execute: exit)
    }

    // MARK: - present / dismiss (Pattern 8 path — adapt to 8a if spike verdict was 8a)

    private func showPopover() {
        guard let controller = widgetController, let panel = controller.window else { return }
        let queue = controller.queueSnapshot
        let content = PopoverContentView(
            queue: queue,
            onRowClick: { [weak self] sid in self?.onRowClick(sessionID: sid) },
            onClearAll: { [weak self] in self?.onClearAll() }
        )
        let pop: NSPopover
        if let existing = popover { pop = existing }
        else {
            pop = NSPopover()
            pop.behavior = .transient
            popover = pop
        }
        let host = NSHostingController(rootView: content)
        pop.contentViewController = host
        pop.contentSize = NSSize(width: 280, height: min(36 * max(1, queue.count), 36 * 8))
        // Anchor: per UI-SPEC, slide-in direction depends on widget corner.
        // Pattern 8 default: anchor relative to the widget panel's contentView, preferredEdge = .minY.
        // For non-NSPopover topologies (8a) the manual position math goes here.
        if let contentView = panel.contentView {
            pop.show(relativeTo: contentView.bounds, of: contentView, preferredEdge: cornerToEdge())
        }
        log.notice("popover shown rows=\(queue.count, privacy: .public)")
    }

    private func dismissPopover() {
        popover?.performClose(nil)
        log.notice("popover dismissed")
    }

    private func cornerToEdge() -> NSRectEdge {
        switch SettingsStore.shared.widgetCorner {
        case .topRight, .topLeft: return .minY     // popover slides down
        case .bottomRight, .bottomLeft: return .maxY  // popover slides up
        }
    }

    // MARK: - actions

    private func onRowClick(sessionID: String) {
        log.notice("[would-jump session=\(sessionID, privacy: .public)]")   // D2-08 verbatim — Phase 3 ITermBridge inherits.
        Task { await SessionRegistry.shared.clearOne(sessionID: sessionID) }
        dismissPopover()
    }

    private func onClearAll() {
        log.notice("popover Clear all")
        Task { await SessionRegistry.shared.clearAll() }
        dismissPopover()
    }
}
```

private var popoverPanel: FloatingWidgetPanel?

private func showPopover() {
    guard let widget = widgetController, let widgetPanel = widget.window else { return }
    let queue = widget.queueSnapshot
    if popoverPanel == nil {
        let p = FloatingWidgetPanel(contentRect: NSRect(x: 0, y: 0, width: 280, height: 200))
        popoverPanel = p
    }
    let p = popoverPanel!
    let content = PopoverContentView(
        queue: queue,
        onRowClick: { [weak self] sid in self?.onRowClick(sessionID: sid) },
        onClearAll: { [weak self] in self?.onClearAll() }
    )
    let host = NSHostingView(rootView: content)
    let h = min(36 * max(1, queue.count), 36 * 8) + (queue.count >= 2 ? 32 : 0)
    host.frame = NSRect(x: 0, y: 0, width: 280, height: h)
    p.contentView = host
    p.setContentSize(host.frame.size)
    // Position: anchor below the widget panel by 4pt, leading-aligned with widget right edge - 280 (TR corner).
    let widgetFrame = widgetPanel.frame
    let store = SettingsStore.shared
    let (px, py): (CGFloat, CGFloat) = {
        switch store.widgetCorner {
        case .topRight:    return (widgetFrame.minX - 280 + 44, widgetFrame.minY - h - 4)
        case .topLeft:     return (widgetFrame.minX, widgetFrame.minY - h - 4)
        case .bottomRight: return (widgetFrame.minX - 280 + 44, widgetFrame.maxY + 4)
        case .bottomLeft:  return (widgetFrame.minX, widgetFrame.maxY + 4)
        }
    }()
    p.setFrameOrigin(NSPoint(x: px, y: py))
    p.orderFront(nil)
    log.notice("popover (sibling-panel) shown rows=\(queue.count, privacy: .public)")
}

private func dismissPopover() {
    popoverPanel?.orderOut(nil)
    log.notice("popover (sibling-panel) dismissed")
}
