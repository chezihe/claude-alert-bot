    1. **Verify xcodegen is installed**: `which xcodegen || (echo "Run: brew install xcodegen" >&2; exit 1)`. If missing, the executor MUST stop and surface the install command — do NOT silently fall back to hand-authoring pbxproj or Xcode GUI.

    2. **Author `project.yml`** at the repo root with this exact contents:
       ```yaml
       name: ClaudeAlertBot
       options:
         deploymentTarget:
           macOS: "14.0"
         createIntermediateGroups: true
         developmentLanguage: en
       targets:
         ClaudeAlertBot:
           type: application
           platform: macOS
           sources:
             - path: App
               name: App
           info:
             path: App/Info.plist
             properties:
               CFBundleIdentifier: com.claudealert.bot
               CFBundleName: Claude Alert Bot
               LSMinimumSystemVersion: "14.0"
               LSUIElement: true
               NSAppleEventsUsageDescription: "Claude Alert Bot needs to switch focus to your iTerm2 tab when a Claude Code task completes."
           settings:
             base:
               PRODUCT_BUNDLE_IDENTIFIER: com.claudealert.bot
               GENERATE_INFOPLIST_FILE: NO
               INFOPLIST_FILE: App/Info.plist
               MACOSX_DEPLOYMENT_TARGET: "14.0"
               SWIFT_VERSION: "5"
           dependencies:
             - target: cab-test
               embed: true
               codeSign: false
           postBuildScripts:
             - name: "Embed Reporter shell script"
               script: |
                 SRC="${SRCROOT}/Reporter/cab-report.sh"
                 DEST="${TARGET_BUILD_DIR}/${UNLOCALIZED_RESOURCES_FOLDER_PATH}/cab-report.sh"
                 if [ -f "$SRC" ]; then
                     mkdir -p "$(dirname "$DEST")"
                     cp "$SRC" "$DEST"
                     chmod +x "$DEST"
                 fi
               shell: /bin/sh
         cab-test:
           type: tool
           platform: macOS
           sources:
             - path: CabTest
               name: CabTest
           settings:
             base:
               GENERATE_INFOPLIST_FILE: NO
               INFOPLIST_FILE: ""
               MACOSX_DEPLOYMENT_TARGET: "14.0"
               SWIFT_VERSION: "5"
       schemes:
         ClaudeAlertBot:
           build:
             targets:
               ClaudeAlertBot: all
               cab-test: all
           run:
             config: Debug
           archive:
             config: Release
       ```
